Tipovi

Konstruktor i this

Class 
class Tocka
{
    public double x;
    public double y;

    public Tocnka(double x, double y)
    {
        this.x = x;
        this.y = y;
    }
}

// stvaranje objekta pomoću konstruktora
Tocka t = new Tocka(2.5, 3.0);
this se koristi za razlikovanje člana klase i parametra konstruktora.
